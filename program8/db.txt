Step 2: Create a text index

db.catalog.createIndex({name:"text",description:"text"
})

Step 3: Build and execute the aggregation pipeline
//stage1


{
    $match: {
        $text: {
            $search: "smartphone"
        }
    }
}



//stage2

{
    $project:{
        _id:0,
        name:1,
        description:1,
        price:1,
        score:{$meta:"textScore"}
    }
}




db.catalog.aggregate([
//Stage 1 : Filter documents using the text index
    {
        $match: {
            $text: {
                $search: "smartphone"
            }
        }
    }
])

//Stage 2: Reshape the output and include the relevance score
{
    $project:{
        _id:0,
        name:1,
        description:1,
        price:1,
        score:{$meta:"textScore"}
    }
}
